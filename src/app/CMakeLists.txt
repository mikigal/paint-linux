configure_file(PaintConfig.h.in ${CMAKE_CURRENT_BINARY_DIR}/PaintConfig.h @ONLY)

# Everything except main() lives in a library so widget tests can link it.
add_library(paint_app STATIC
    ActionRegistry.cpp
    ActionRegistry.h
    MainWindow.cpp
    MainWindow.h
    canvas/CanvasView.cpp
    canvas/CanvasView.h
    canvas/SidePanel.cpp
    canvas/SidePanel.h
    panels/LayersDock.cpp
    panels/LayersDock.h
    ribbon/ColorsPanel.cpp
    ribbon/ColorsPanel.h
    ribbon/PaintRibbon.cpp
    ribbon/PaintRibbon.h
    ribbon/PreviewMenu.cpp
    ribbon/PreviewMenu.h
    ribbon/Ribbon.cpp
    ribbon/Ribbon.h
    ribbon/RibbonButton.cpp
    ribbon/RibbonButton.h
    ribbon/RibbonHost.cpp
    ribbon/RibbonHost.h
    ribbon/ShapeGallery.cpp
    ribbon/ShapeGallery.h
    ui/BrushPreview.cpp
    ui/BrushPreview.h
    ui/Icons.cpp
    ui/Icons.h
    ui/Names.cpp
    ui/Names.h
    widgets/ZoomControls.cpp
    widgets/ZoomControls.h
)

target_include_directories(paint_app PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(paint_app
    PUBLIC paint_core Qt6::Widgets Qt6::Svg
    PRIVATE paint_warnings)

# Paint-style icons generated by tools/icons/build_icons.py, available as :/paint/icons/<name>.svg.
file(GLOB paint_icon_files CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/resources/icons/*.svg)
qt_add_resources(paint_app "paint_icons"
    PREFIX "/paint/icons"
    BASE ${CMAKE_SOURCE_DIR}/resources/icons
    FILES ${paint_icon_files})

if(PAINT_AUTOMATION)
    target_sources(paint_app PRIVATE
        automation/AutomationServer.cpp
        automation/AutomationServer.h)
    target_compile_definitions(paint_app PUBLIC PAINT_AUTOMATION_ENABLED)
    target_link_libraries(paint_app PUBLIC Qt6::Network Qt6::Test Qt6::GuiPrivate)
endif()

qt_add_executable(${PAINT_EXECUTABLE} main.cpp)
target_link_libraries(${PAINT_EXECUTABLE} PRIVATE paint_app paint_warnings)

install(TARGETS ${PAINT_EXECUTABLE} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
